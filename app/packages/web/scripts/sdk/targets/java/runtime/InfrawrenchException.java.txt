package com.infrawrench.sdk;

/**
 * Base class for every failure this SDK raises.
 *
 * <p>Unchecked on purpose. A checked hierarchy would put {@code throws} on all
 * of the generated methods, which buys nothing — there is no meaningful local
 * recovery from "the network is down" at each of a hundred-odd call sites — and
 * would make adding a new failure mode a source-breaking change for every
 * caller. Catch this type when you want "anything the SDK can go wrong with".
 */
public abstract class InfrawrenchException extends RuntimeException {

  private static final long serialVersionUID = 1L;

  protected InfrawrenchException(String message) {
    super(message);
  }

  protected InfrawrenchException(String message, Throwable cause) {
    super(message, cause);
  }
}
